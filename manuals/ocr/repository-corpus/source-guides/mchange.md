TTL MEMORY CHANGE ROUTINE

	OPT PAG

	PAG
*************************************
* MEMORY EXAMINE AND CHANGE UTILITY *
*************************************
*	BY MICK ALLEN	*
*************************************

* FLEX SYSTEM EQUATES

ESCRET EQU $CC16
WARMS	EQU $CD03
GETCHR EQU $CD15
PUTCHR EQU $CD18
INBUFF EQU $CD1B
PSTRNG EQU $CD1E
CLASS	EQU $CD21
PCRLF	EQU $CD24
OUTHEX EQU $CD3C
GETHEX EQU $CD42
OUTADR EQU $CD45
STAT	EQU $CD4E

	ORG $C100


MEMCHA BRA MEMCHG
VER	FCB 1 VERSION
TEMP	FDB 0 TEM STORAGE


MEMCHG JSR PCRLF START ON NEW LINE
	LDX #MESG1 POINT TO BANNER
	JSR PSTRNG PRINT IT
	JSR PCRLF MISS A LINE
	BRA MEMC2 SKIP OVER ERROR MESSAGE
*
* WARN OPERATOR OF FUNCTION ERROR
*
ERROR	LDX #ERRMSG POINT TO ERROR MESSAGE
	JSR PSTRNG PRINT IT
*
* ASK FOR COMMAND FUNCTION
*
MEMC2	LDX #MESG2 POINT TO FUNCTIONS
	JSR PSTRNG PRINT IT
*
* GET FUNCTION CODE LETTER AND CHECK
*
	JSR GETCHR GET CHARACTER FROM KBD
	CMPA #'S IS IT "S" ?
	LBEQ WARMS YES, GO TO FLEX
	CMPA #'N IS IT "N" ?
	BNE ERROR NO, TELL USER
	BRA ADDRIN SKIP OVER ERROR PROMPT
*
* WARN OPERATOR OF ADDRESS ERROR
*
ERROR2 LDX #ADRERR POINT TO ADDRESS ERROR MESSAGE
	JSR PSTRNG PRINT IT
*
* GET START ADDRESS
*
ADDRIN LDX #MESG3 POINT TO 'ADDR = '
	JSR PSTRNG PRINT IT
	JSR INBUFF GET INPUT INTO BUFFER
	JSR GETHEX CONVERT TO HEX
	BCS ERROR2 IF ERROR TELL USER
	TFR X,Y PUT ADDRESS IN Y
MEMC3	STY TEMP SAVE ADDRESS FOR PRINTING
*
* PRINT ADDRESS
*
	LDX #MESG3 POINT TO 'ADDR = '
	JSR PSTRNG PRINT IT
	LDX #TEMP POINT TO ADDRESS
	BSR OUT4H PRINT IT
	BSR OUT1S LEAVE A SPACE
*
* PRINT DATA
*
	LDY TEMP GET ADDRESS INTO Y
	BSR OUT2HS PRINT DATA & SPACE
*
* GET INPUT FROM USER
*
	BSR BYTE GET NEW DATA
	BVC CHANGE IF OK, CHANGE IN RAM
	CMPA #$08 IF NOT, IS IT BS ?
	BEQ MEMC3 Y, RE-DO THIS ADDRESS
	CMPA #$1B IS IT ESC ?
	BEQ MEMCHG Y, RE-PROMPT
	CMPA #$5E '^' IS IT UP ARROW ?
	BEQ BACK Y, RE-DO LAST
	CMPA #$0D IS IT CR ?
	BNE FWARD NO, THEN DO NEXT
	BRA MEMCHG RE-PRINT BANNER ETC

OUT2HS LDA ,Y GET DATA
	BSR OUT2H OUTPUT A AS 2 HEX DIGITS
OUT1S	LDA #$20 ASCII CODE FOR SPACE
OUTCH	JSR PUTCHR OUTPUT THRO' FLEX
	RTS

CHANGE STA ,Y
	CMPA ,Y
	BEQ FWARD
	BSR OUT1S
	LDX #MESG4 POINT TO MESSAGE
	JSR PSTRNG PRINT IT
FWARD	LEAY 1,Y
	BRA MEMC3

BACK	LEAY -1,Y
	BRA MEMC3

OUT4H	LDA ,X+ GET MS BYTE
	BSR OUT2H PRINT AS 2 HEX DIGITS
	LDA ,X+ GET LS BYTE

OUT2H	PSHS A SAVE ON STACK
	LSRA CONVERT TOP 4 BITS TO ASCII
	LSRA
	LSRA
	LSRA
	BSR OUTHL
	PULS A RESTORE DATA IN A
OUTHL	ANDA #$0F MASK OFF TOP 4 BITS
	ADDA #'0 CONVERT HEX TO ASCII
	CMPA #$39 IS IT MORE THAN "9" ?
	BLE OUTCH NO, THEN PRINT IT
	ADDA #$07 YES, THEN MAKE INTO LETTER (A-F)
	BRA OUTCH AND OUTPUT
*
* GET A BYTE FROM KEYBOARD (2 DIGITS)
*
BYTE	BSR INHEX GET A HEX DIGIT
	BVS NOBYTE EXIT IF INVALID
	ASLA MOVE INTO TOP 4 BITS
	ASLA
	ASLA
	ASLA
	TFR A,B SAVE A
	BSR INHEX GET ANOTHER HEX DIGIT
	BVS NOBYTE EXIT IF INVALID
	PSHS B SAVE TOP 4 BITS ON STACK
	ADDA ,S+ ADD TOP BITS & TIDY STACK
NOBYTE RTS

*
* GET A HEX DIGIT (SET V IF INVALID KEY)
*
INHEX	JSR GETCHR
	CMPA #'0 IS IT > or = " 0 "
	BCS NOTHEX IF < IT'S NOT HEX
	CMPA #'9 IS IT < or = " 9 "
	BHI INHEXA IF > MAYBE IT'S A-F
	SUBA #$30 CONVERT TO HEX
	RTS AND RETURN
INHEXA CMPA #'A IS IT > or = " A " ?
	BCS NOTHEX IF < IT'S NOT HEX
	CMPA #'F IS IT < or = " F " ?
	BHI NOTHEX IF > IT'S NOT HEX
	SUBA #$37 CONVERT TO HEX
	RTS
NOTHEX ORCC #$02 IT'S MORE, SO SET "V"
	RTS AND RETURN


MESG1	FCC ' Memory Examine & Change',$04
MESG2	FCC ' (N)ew address, or (S)top',$0D,$0A,0,0,0,0
	FCC ' Enter command letter : ',$04
MESG3	FCC ' Addr.= $',$04
MESG4	FCC 'Data read back is not equal to data written!',$04
ERRMSG FCC ' INVALID COMMAND!',$04
ADRERR FCC ' INVALID ADDRESS!',$04

	END MEMCHA
