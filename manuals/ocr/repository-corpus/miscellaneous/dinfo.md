TTL	DISPLAY DISK INFORMATION - 10,25,81
*
************************************************
*
* THIS PROGRAM GIVES THE FOLLOWING INFORMATION:
*
* DISK DRIVE NUMBER
* DISKETTE NAME + EXTENSION
* DISKETTE VOLUMNE NUMBER
* DISKETTE CREATION DATE
* NUMBER OF TRACKS & SECTORS ON DISKETTE
* NUMBER OF ENTRIES IN THE DIRECTORY
* NUMBER OF SECTORS USED ON DISKETTE
* NUMBER OF SECTORS REMAINING ON DISKETTE
* NEXT FREE SECTOR IN FREE CHAIN
* WHETHER DISK IS LINKED TO BOOT
* DISK LINK ADDR
* DISK LINK FILE NAME
* FILE ATTRIBUTES COUNT
*
************************************************
*

* FLEX SUBROUTINE LINKAGES

*
* FCB INDICES (X)

*

*

	ORG	$C100

START1 LEAU	PEND,PCR	WORK SPACE
	LDB	#32
STA	CLR	B,U
	DECB
	BPL	STA
	LDD	ERF
	LBSR	PUTTWO		HOME AND CLEAR SCREEN
	LEAX	TITLE,PCR	PRINT NAME OF PROGRAM
	JSR	[PSTRNG,PCR]
	JSR	[PCRLF,PCR]	SPACE DOWN DISPLAY
	CLR	DRV,U
SET DEFAULTS
	JSR	[NXTCHR,PCR]	GET DRIVE NUMBER IF THERE
	CMPA	#$D
CHECK FOR NO DRIVE
	BEQ	CONT
	CMPA	#'3
MAX 3 ALLOWED
	BLS	CONT1
	CMPA	#'0
	BLO	CONT
CONT1	SUBA	#'0
	STA	DRV,U
CONT	LEAX	DRIVMG,PCR
	JSR	[PSTRNG,PCR]	REPORT DRIVE NUMBER
	LDA	DRV,U
	ADDA	#'0
	JSR	[PUTCHR,PCR]
	LBSR	READSY
	BSR	DOSYS
DO SYSTEM INFO DATA
	LBSR	READBT		READ BOOT SECTOR
	LBSR	DOBOOT		DO BOOT DATA
	LBSR	DODIR
DO DIRECTORY DATA
	LBRA	EXIT
*
* PRINT DISK NAME AND EXTENSION
*
DOSYS	LEAX	DNAMMG,PCR
	JSR	[PSTRNG,PCR]
	LDX	#FCB+DATINX+SDNAME	POINT TO DISK NAME
	LDB	#8
# OF LETTERS IN NAME
NAME1	LDA	,X+
PRINT DISK NAME
	BEQ	NAME1A
	JSR	[PUTCHR,PCR]
NAME1A DECB
	BNE	NAME1
	LDA	#'.
	JSR	[PUTCHR,PCR]
	LDB	#3
# OF LETTERS IN EXTENSION
NAME2	LDA	,X+
PRINT EXTENSION NAME
	BEQ	NAME2A
	JSR	[PUTCHR,PCR]
NAME2A DECB
	BNE	NAME2
*
* PRINT DISKETTE VOLUME NUMBER
*
	LEAX	DVOL,PCR	PRINT VOLUMNE NUMBER
	JSR	[PSTRNG,PCR]
	LDX	#FCB+DATINX+VOLUMN
	JSR	[OUTDEC,PCR]	PRINT IN DECIMAL
*
* PRINT DISKETTE INITIALIZATION DATE
*
	LEAX	DDATE,PCR
	JSR	[PSTRNG,PCR]
	LDY	#FCB+DATINX+INDATE
	LEAX	MONTAB,PCR
	LDB	,Y
	DECB
	LDA	#3
	MUL
	ABX	POINT TO MONTH STRING
	LDB	#3
DATE1	LDA	,X+
	JSR	[PUTCHR,PCR]
	DECB
	BNE	DATE1
	LDA	#$20
	JSR	[PUTCHR,PCR]
	CLRB
	CLR	Y
	LEAX	Y	PRINT DAY OF MONTH
	JSR	[OUTDEC,PCR]
	LEAY	1,Y
	CLR	Y
	LEAX	Y
	LDA	#',
	JSR	[PUTCHR,PCR]
	LDD	#$3139		'19
	LBSR	PUTTWO
	CLRB
	JSR	[OUTDEC,PCR]	PRINT YEAR
*
* PRINT NUMBER OF TRACKS & SECTORS ON DISK
*
	LEAY	1,Y
	CLR	Y
	LDD	1,Y
	MUL
	STD	MAXALL,U	SAVE DISK SIZE
	INC	1,Y
	LEAX	DSIZE1,PCR
	JSR	[PSTRNG,PCR]
	LEAX	Y
	CLRB
	JSR	[OUTDEC,PCR]	PRINT IN DECIMAL
	LEAY	1,Y
	CLR	Y
	LEAX	DSIZE2,PCR
	JSR	[PSTRNG,PCR]
	LEAX	Y
	CLRB
	JSR	[OUTDEC,PCR]
*
* PRINT NEXT TRACK & SECTOR FREE LINK
*
	LEAX	DNXSEC,PCR
	JSR	[PSTRNG,PCR]
	LEAX	TRKMSG,PCR
	JSR	[PSTRNG,PCR]
	LDY	#FCB+DATINX+FREECH-1	POINT TO FREE CHAIN
	CLR	Y
	LEAX	Y
	CLRB
	JSR	[OUTDEC,PCR]
	LEAX	SECMSG,PCR
	JSR	[PSTRNG,PCR]
	LEAY	1,Y
	CLR	Y
	LEAX	Y
	CLRB
	JMP	[OUTDEC,PCR]
*
DOBOOT LEAX	DLINK,PCR
	JSR	[PSTRNG,PCR]
	LDX	#FCB+DATINX+LNKLOC
	TST	X
	BEQ	DONO	CANNOT BE ZERO IF LINKED
	COM	LFLAG,U
	LEAX	YESMSG,PCR
DOB2	LDB	#3
DOB1	LDA	,X+
	JSR	[PUTCHR,PCR]
	DECB
	BNE	DOB1
	LDD	#$2020
	LBSR	PUTTWO
	LDA	#'(
	JSR	[PUTCHR,PCR]
	LDX	#FCB+DATINX+5
	LDD	X
	STD	LINKAD,U
SAVE LINK FILE ADDR
	JSR	[OUTHEX,PCR]	PRINT LINK
	LDA	#',
	JSR	[PUTCHR,PCR]
	LEAX	1,X
	JSR	[OUTHEX,PCR]
	LDA	#')
	JSR	[PUTCHR,PCR]
	LDD	#$2020
	LBRA	PUTTWO
*
DONO	LEAX	NOMSG,PCR
	BRA	DOB2
*
DODIR	LDX	#FCB
	LDA	#OPNDIR
	STA	FUNC,X
	LDA	DRV,U
	STA	DRIVE,X
	JSR	[FMS,PCR]
	LBNE	DOERR
	LDY	#$FFFF
	STY	LOWEST,U
	LEAY	1,Y
	STY	BIGGST,U
DOLOOP LDA	#GTINFO	READ A DIR ENTRY
	STA	FUNC,X
	JSR	[FMS,PCR]
	LBNE	CHKEND
	TST	DNAME,X	TEST FOR DELETED OR UNUSED ENTRY
	BEQ	DOLOOP
	BMI	DOLOOP
	TST	LFLAG,U	TEST FOR LINKED
	BPL	CONT2
IF NO
	LDD	SDADDR,X	IF YES, CHECK FOR MATCH
	CMPD	LINKAD,U
	BNE	CONT2
	LDB	#8
IF MATCH, PRINT NAME
	LEAY	4,X
LNAME	LDA	,Y+
	BEQ	LNAM1
	JSR	[PUTCHR,PCR]
LNAM1	DECB
	BNE	LNAME
	LDA	#'.
	JSR	[PUTCHR,PCR]
	LDB	#3
LNAME1 LDA	,Y+
	BEQ	LNAM2
	JSR	[PUTCHR,PCR]
LNAM2	DECB
	BNE	LNAME1
CONT2	LDA	DATTR,X	FILE FILE ATTRIBUTES
	PSHS	A
	ANDA	#$80

	BNE	IWRT
IC1	LDA	S
	ANDA	#$40

	BNE	IDEL
IC2	LDA	S
	ANDA	#$20

	BNE	IREAD

	ANDA	#$10

	BEQ	IC4
	LDD	CAT,U
	ADDD	#1
	STD	CAT,U
	BRA	IC4

	ADDD	#1
	STD	WRT,U
	BRA	IC1
IDEL	LDD	DEL,U
	ADDD	#1
	STD	DEL,U
	BRA	IC2

	ADDD	#1
	STD	READ,U
	BRA	IC3
*
IC4	LDD	DFSIZE,X	FIND LARGEST & SMALLEST FILES
	CMPD	BIGGST,U
	BLS	PAST
	STD	BIGGST,U

	BHS	PAST1
	STD	LOWEST,U

	STD	SECUSE,U
	LDD	FUSED,U	NUMBER OF FILES ON DISK
	ADDD	#1
	STD	FUSED,U
	LBRA	DOLOOP
*

	CMPA	#FEOF
	LBNE	DOERR
	LDD	MAXALL,U
	SUBD	SECUSE,U
	STD	SECUNU,U	SECTOR COUNT UNUSED
	LEAX	DSECUS,PCR	TELL HOW MANY SECTORS USED
	JSR	[PSTRNG,PCR]
	LEAX	SECUSE,U
	CLRB
	JSR	[OUTDEC,PCR]
	LEAX	DSECUN,PCR	TELL MANY MANY SECTORS UNUSED
	JSR	[PSTRNG,PCR]
	LEAX	SECUNU,U
	CLRB
	JSR	[OUTDEC,PCR]
	LEAX	BIGMSG,PCR
	JSR	[PSTRNG,PCR]
	LEAX	BIGGST,U
	CLRB
	JSR	[OUTDEC,PCR]
	LEAX	SMAMSG,PCR	INDICATE SMALLEST FILE SIZE
	JSR	[PSTRNG,PCR]
	LEAX	LOWEST,U
	CLRB
	JSR	[OUTDEC,PCR]
	LEAX	ATTMSG,PCR
	JSR	[PSTRNG,PCR]
	LEAX	ATTD,PCR
	JSR	[PSTRNG,PCR]	SHOW ATTRIBUTES PER NUMBER
	LEAX	DEL,U
	CLRB
	JSR	[OUTDEC,PCR]
	LEAX	ATTW,PCR
	JSR	[PSTRNG,PCR]
	LEAX	WRT,U
	CLRB
	JSR	[OUTDEC,PCR]
	LEAX	ATTR,PCR
	JSR	[PSTRNG,PCR]
	LEAX	READ,U
	CLRB
	JSR	[OUTDEC,PCR]
	LEAX	ATTC,PCR
