WARMAD EQU $AD03 WARM START
INEAD EQU $AD15 INPUT 1 CHAR TO A
OUTEAD EQU $AD18 OUTPUT 1 CHAR FROM A
PCRLAD EQU $AD24 PRINT CRLF
ADDBAD EQU $AD36 ADD B TO X
RPTEAD EQU $AD3F REPORT ERROR
FCLSAD EQU $B403 FMS CLOSE
FMSAD EQU $B406 FMS CALL
WASNAD EQU $AC0C WORK DRIVE
ESCRAD EQU $AC16 ESCAPE RETURN
OPNINP EQU $01 OPEN INPUT
OPNOUT EQU $02 OPEN OUTPUT
CLSFIL EQU $04 CLOSE FILE
READSS EQU $09 READ SINGLE SECTOR
**
CR EQU $0D ASCII CR
EOT EQU $04 ASCII EOT
CLRSCN EQU $1A80 CLEAR SCREEN
**
START BRA BEGIN STARTING ADDR
VN FCB $01 VERSION NUMBER
**
JMPBEG EQU * FIRST FLEX JUMP
WARMS JMP WARMAD WARM START
INEEE JMP INEAD INPUT 1 CHAR TO A
OUTEE JMP OUTEAD OUTPUT 1 CHAR FROM A
PCRLF JMP PCRLAD PRINT CRLF
ADDBX JMP ADDBAD ADD B TO X
RPTERR JMP RPTEAD REPORT ERROR
FMSCLS JMP FCLSAD FMS CLOSE
FMS JMP FMSAD FMS CALL
JMPEND EQU * LAST FLEX JUMP
FDBBEG EQU * FIRST FLEX FDB
WASN FDB WASNAD WORK DRIVE
ESCRTN FDB ESCRAD ESCAPE RETURN
FDBEND EQU * LAST FLEX FDB

DISKD JSR XCRLF INPUT FILE
 LDX #DISIM
 JSR XPDAT
 LDX #FCBI
 STX BADDR
 JSR SCNFLX GET INPUT FILE NAME
 CMPB #$1F
 BEQ DISKD RETRY IF NO INPUT
 LDX #FCBI
 TST $0C,X CHK EXTENSION
 BNE DIKEI
 LDAA #$54 T SET EXTENSION
 STAA $0C,X
 LDAA #$58 X
 STAA $0D,X
 LDAA #$54 T
 STAA $0E,X
DIKEI LDAA #OPNINP OPEN FILE
 STAA $00,X
 JSR FMS
 BEQ DIKNI CHK ERROR
 JMP RPTERR
DIKNI JSR XCRLF CONTROL FILE
 LDX #DISCM
 JSR XPDAT
 LDX #FCBC
 STX BADDR
 JSR SCNFLX GET CONTROL FILE NAME
 CMPB #$1F
 BEQ DIKNI RETRY IF NO INPUT
 LDX #FCBC
 TST $0C,X CHK EXTENSION
 BNE DIKEC
 LDAA #$54 T SET EXTENSION
 STAA $0C,X
 LDAA #$58 X
 STAA $0D,X
 LDAA #$54 T
 STAA $0E,X
DIKEC LDAA #OPNINP OPEN FILE
 STAA $00,X
 JSR FMS
 BEQ DIKNC CHK ERROR
 JMP RPTERR
DIKNC JSR XCRLF OUTPUT FILE
 LDX #DISOM
 JSR XPDAT
 LDX #FCBO
 STX BADDR
 JSR SCNFLX GET OUTPUT FILE NAME
 CMPB #$1F
 BEQ DIKNC RETRY IF NO INPUT
 LDX #FCBO
 LDAA $04,X CHK FILE NAME
 CMPA #$41
 BMI DIKNC
 CMPA #$5A
 BHI DIKNC
 TST $0C,X CHK EXTENSION
 BNE DIKEO
 LDAA #$54 T SET EXTENSION
 STAA $0C,X
 LDAA #$58 X
 STAA $0D,X
 LDAA #$54 T
 STAA $0E,X
DIKEO LDAA #OPNOUT OPEN FILE
 STAA $00,X
 JSR FMS
 BEQ DIKNO CHK ERROR
 JMP RPTERR
DIKNO BRA CONFIL READ CONTROL FILE
DIKPR JMP PROCES PROCESS INPUT FILE
DIKOC LDX #FCBO CLOSE OUTPUT FILE
 LDAA #CLSFIL
 STAA $00,X
 JSR FMS
 BEQ EXITS
 JSR RPTERR
EXITS JSR FMSCLS CLOSE FILES
 JMP WARMS RETURN TO FLEX

 LDX CONTAB
CONFER STX TABS
CONFRE STX SADDR

 CMPA #CR CHK CR

CONFX1 INX GET NEXT CHAR
 STX SADDR

 BRA CONFRE
CONFNC LDX SADDR STORE IT
 STAA $00,X
 CMPA DELIM

CONFX2 INX GET NEXT CHAR

CONRED LDX #FCBC READ CONTROL FILE
 JSR FMS
 BNE CONRER
 CMPA #CR
 BEQ CONROK
 CMPA #$1F
 BLS CONRED
 BRA CONROK
CONRER LDAA $01,X
 CMPA #$08
 BEQ CONRNO
 JSR RPTERR
CONRNO LDAA #EOT EOF

 STAA $00,X

PROCES CLR BUFFER+1 PROCESS DATA
 LDX BUFFER
 STX ENDR INIT END POINTER
INITER LDX ENDR RESET START POINTER
 STX STAR
BACKUP LDX STAR RESET BUFFER POINTER
 STX BUFF
GNCHAR BSR INCHAR GET A CHAR
CHEKER JSR MATCH CHK MATCH
 TSTB
 BEQ GNCHAR PARTIAL MATCH
 BPL SUBMAT FULL MATCH
 LDX STAR NO MATCH
 LDAA $00,X OUT CHAR AT START
 BSR OUTCHR
 INC STAR+1 BUMP POINTER
 BRA BACKUP
SUBMAT LDX TABS OUT NEW STRING
SUBBER INX
 LDAA $00,X
 CMPA DELIM
 BEQ INITER
 STX BADDR
 BSR OUTCHR
 LDX BADDR
 BRA SUBBER
**
OUTCHR CMPA #EOT OUTPUT A CHAR
 BNE OUTCHE CHK EOF
 JMP DIKOC CLOSE FILES AND EXIT
OUTCHE LDX #FCBO OUT CHAR
 JSR FMS
 BEQ OUTCHX
 JSR RPTERR
 JMP DIKOC EXIT ON ERROR
OUTCHX RTS
**
INCHAR LDX ENDR INPUT A CHAR
 CPX BUFF
 BNE INCHAB
 LDX #FCBI IN CHAR
 JSR FMS
 BNE INCHEN
 CMPA #CR CHK CR
 BEQ INCHNE
 CMPA #$1F
 BLS INCHAR
 BRA INCHNE
INCHEN LDAA $01,X CHK EOF
 CMPA #$08
 BEQ INCHEF
 JSR RPTERR
INCHEF LDAA #EOT EOF
INCHNE LDX ENDR STORE IT
 STAA $00,X
 INC ENDR+1
INCHAB INC BUFF+1
 RTS
**
MATCH CLRB CHK FOR MATCH
 COMB ASSUME NO MATCH
 LDX CONTAB RESET POINTER
MATGGL STX RADDR
MATGDL LDAA $00,X DELIMITER
 CMPA #EOT CHK EOF
 BEQ MATCHX
 STAA DELIM
 LDX STAR RESET BUFFER POINTER
 STX STAT
MATCMS LDX RADDR CHK OLD STRING
 INX
 STX RADDR
 LDAA $00,X
 CMPA DELIM
 BEQ MATFAL
 LDX STAT
 CMPA $00,X
 BNE MATFAL
 INC STAT+1
 LDX STAT
 CPX BUFF
 BNE MATCMS
 LDX RADDR CHK FULL/PART MATCH
 INX
 STX RADDR
 CLRB ASSUME PART MATCH
 LDAA $00,X
 CMPA DELIM
 BNE MATFAL
 INCB FULL MATCH
 STX TABS POINT TO NEW STRING
 BRA MATCHX
MATFAL LDX RADDR NO MATCH HERE
MATFLI INX
 LDAA $00,X
 CMPA #CR CHK FOR CR
 BNE MATFLI
 INX
 BRA MATGGL
MATCHX RTS EXIT
