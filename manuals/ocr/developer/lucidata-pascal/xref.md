	VAR
	K : INTEGER ; ID : ALFA ; ENDID : BOOLEAN ;
	BEGIN

PROCEDURE LOCATE ( THISONE,SIDE : INTEGER ) ;
	VAR
	J,THATONE : INTEGER ;
	BEGIN

	END;
	END; (*** OF LOCATE ***)
