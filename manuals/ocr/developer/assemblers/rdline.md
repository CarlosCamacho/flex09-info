TTL RDLINE
 OPT PAG
 PAG
*
* THIS ROUTINE GETS A LINE OF SOURCE FROM
* INBUFF AND RETURNS THE ADDRESS OF THE
* LINE IN CULINE.
* CUCHAR := ADDRESS OF FIRST CHARACTER
*
RDLINE TST MACFLG MODE = MACRO?
 BEQ RDLINA NO
 JSR RDMAC EXPAND MACRO
 TST MACFLG IS MACRO FULLY EXPANDED?
 BEQ RDLINA YES
 RTS
*
RDLINA LDX #INLINE
 STX CUCHAR
 STX CULINE
RDL1 JSR GETB
 BCC RDL1A
 LDS #$A042 FLUSH STACK - EOF
 JMP POEND0
*
RDL1A CMPA #$0A LF?
 BEQ RDL1 YES
 CMPA #$00 NULLS?
 BEQ RDL1 YES
 CPX #INLINE+79 LINE TO LONG?
 BEQ RDL2 YES
 STAA 0,X STORE CHARACTER
 INX
 BRA RDL3
RDL2 LDAB #$0D TRUNCATE LINE
 STAB 0,X
RDL3 CMPA #$0D IS IT CR?
 BNE RDL1 NO
 RTS
*
* EXPAND MACRO CALL
*
RDMAC LDX MACPTR
 LDAA 0,X GET CHARACTER
 CMPA #$17 ETB?
 BNE RDMAC1 NO
 DEC MACFLG DEC MODE COUNT
 BEQ RDMAC0 NO MORE MACROS
*
* PULL UP LAST MACRO STACKED
*
 JSR MACPUL
 BRA RDMAC
RDMAC0 RTS
*
RDMAC1 LDX #MACLIN POINT TO MACRO EXPAND AREA
 STX CULINE INIT
 STX CUCHAR INIT
 STX MCLPTR INIT
RDMAC2 LDX MACPTR POINT TO MACRO DEFINITION
 LDAA 0,X
 INX
 STX MACPTR
 CMPA #'& MACRO PARAMETER?
 BEQ RDMAC3 YES
 LDX MCLPTR POINT TO MACRO LINE
 STAA 0,X MOVE CHAR TO MACLIN
 INX
 STX MCLPTR SAVE POINTER
 CPX #MACLIN+71 OVERFLOW?
 BEQ RDMERR YES - ERROR
 CMPA #$0D END OF LINE?
 BNE RDMAC2 NO
 RTS YES - EXIT
*
* SUBSTITUTE POSITIONAL PARAMETERS
*
RDMAC3 LDAB 0,X GET POSITIONAL # OF PARAM
 SUBA #$2F CONVERT TO BINARY
 INX SKIP OVER POSITION #
 STX MACPTR
 LDX #MACPAR POINT TO PARAMS FROM CALL
*
* SCAN OVER PARAMETERS
*
RDMAC6 STX MACSAV SAVE
RDMAC4 LDAA 0,X GET A CHARACTER
 INX
 CMPA #', END OF PARAMTER?
 BEQ RDMAC7 YES
 CMPA #$0D END OF LINE?
 BNE RDMAC4 NO
RDMAC7 DECB FOUND PARAMTERE?
 BNE RDMAC6 NO
 LDX MACSAV POINT TO PARAMETER
 LDAA 0,X GET CHARACTER
 INX
 STX MACSAV SAVE POINTER
RDMAC5 LDX MCLPTR POINT TO MACLIN
 STAA 0,X MOVE CHARACTER
 INX
 STX MCLPTR SAVE POINTER
 CPX #MACLIN+71 OVERFLOW?
 BEQ RDMERR YES - ERROR
*
 LDX MACSAV POINT TO MACRO PARAMETER
 LDAA 0,X GET NEXT CHARACTER
 INX
 STX MACSAV SAVE POINTER
 CMPA #', END OF PARAMETER?
 BEQ RDMAC2 YES
 CMPA #$0D END OF LINE?
 BNE RDMAC5 NO
 BRA RDMAC2
RDMERR LDAA #$0D END OF LINE
 STAA 0,X
 LDX #$0203 ERROR MESSAGE
 JSR PRINTE PRINT ERROR
 BRA RDMAC2
*
* PUSH A MACRO ONTO THE MACRO STACK
*
MACPSH STX MXSAV1 SAVE X-REGISTER
 STS STKSAV SAVE STACK POINTER
 LDS MSTKPT LOAD MACRO STACK POINTER
*
* PUSH MCLPTR,MACSAV AND MACPTR ONTO STACK
*
 LDX #MCLPTR+1
 LDAB #6
MPSH1 LDAA 0,X GET A BYTE
 DEX
 STX MXSAV2 SAVE POINTER
 TSX X := STKPTR+1
 DEX
 CPX MACEND END OF STACK?
 BEQ MPSH5 YES - ERROR
 LDX MXSAV2 RESTORE POINTER
 PSHA PUSH A BYTE ONTO STACK
 DECB	ALL DONE?
 BNE MPSH1 NO
*
* PUSH MACPAR IN REVERSE ORDER
*
 LDX #MACPAR
MPSH2 LDAA 0,X FIND EOL
 CMPA #$0D EOL?
 BEQ MPSH3 YES
 INX
 BRA MPSH2
*
MPSH3 LDAA 0,X GET A BYTE
 STX MXSAV2 SAVE POINTER
 TSX X := STKPTR+1
 DEX
 CPX MACEND END OF STACK?
 BEQ MPSH5 YES - ERROR
 LDX MXSAV2 RESTORE POINTER
 PSHA PUT ON STACK
 DEX POINT TO NEXT CHAR
 CPX #MACPAR-1 ALL DONE?
 BNE MPSH3 NO
 STS MSTKPT SAVE STACK POINTER
 LDS STKSAV RESTORE STACK
 LDX MXSAV1 RESTORE X-REGISTER
 RTS
*
* MACRO NESTING OVERFLOW ERROR
*
MPSH5 LDS MACSTK FLUSH STACK
 STS MSTKPT
 LDS STKSAV
 LDX #$0251 ERROR NUMBER
 JSR PRINTE
 LDX MXSAV1 RESTORE INDEX
 CLR MACFLG MODE <> MACRO
 RTS
*
* PULL A MACRO FROM STACK
*
MACPUL STX MXSAV1 SAVE INDEX
 STS STKSAV SAVE STACK POINTER
 LDS MSTKPT LOAD MACRO STACK POINTER
*
* PULL MACPAR FROM STACK
*
 LDX #MACPAR
MPUL1 PULA PULL A CHARACTER
 STAA 0,X SAVE IN MACPAR
 INX
 CMPA #$0D END OF LINE?
 BNE MPUL1 NO
*
* PULL MACPTR,MACSAV AND MCLPTR
*
 LDX #MACPTR
 LDAB #6
MPUL2 PULA PULL A CHARACTER
 STAA 0,X SAVE
 INX
 DECB
 BNE MPUL2 NOT DONE
 STS MSTKPT SAVE MACRO STACK POINTER
 LDS STKSAV RESTORE STACK POINTER
 LDX MXSAV1 RESTORE INDEX
 RTS
