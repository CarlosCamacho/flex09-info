* PROCESS FDB PSEUDO OP
*
POFDB JSR ADRINT
 INC FCBFLG
 JSR NXTOK
 CMPB #$0D EOL?
 BNE FDBB NO
FDBA LDX #$0216 ERROR
 JSR PRINTE
 BRA FDBC
*
FDBB JSR NSEVL EVALUATE OPERAND
 JSR P2ERR
 INC LCN
 INC LCN
 TST PASS
 BEQ FDBD IF PASS 1
 LDAB ADR1
 JSR OUTBIN
 LDAB ADR2
 JSR OUTBIN
 TST RELFLG RELOC?
 BEQ FDCF NO
 LDAB #'R YES
 BRA FDCG
*
FDCF TST CMNFLG COMMON?
 BEQ FDCE NO
 LDAB #'M YES
FDCG JSR OUTBNR OUTPUT "R" OR "M"
FDCE INC MCOUNT
 INC MCOUNT
 COM POP
FDBC JSR PRINTL
FDBD JSR LCLCN

* PROCESS THE MAC PSEUDO OP
*
POMAC JSR ADRINT
 JSR PRINTL
 CLR CMFLG
 CLR MACERR
 TST PASS
 BNE POMAC2 PASS2
 TST LBFLG LABELED?
 BNE POMAC1 YES - GOOD
 COM MACERR SET ERROR FLAG
 LDX #$0226
 JSR PRINTE
 BRA POMAC
*
POMAC1 LDX SYMPTR
 LDAA #$20
 STAA 8,X SET MACRO FLAG IN SYMTAB
 LDAA MACPTR
 STAA 6,X
 LDAA MACPTR+1
 STAA 7,X
 JSR NXTOK
 LDX DESCRA
 LDAA 0,X
 CMPA #'C
 BNE *+5
 COM CMFLG
POMAC2 JSR RDLINA GET NEXT LINE
 LDX LNUM
 INX
 STX LNUM
 JSR PRINTL
 LDX CULINE
 LDAA 0,X
 CMPA #'* COMMENT?
 BNE POMAC5 NO
 TST CMFLG SAVE COMMENTS?
 BEQ POMAC2 NO
 JSR MACMOV YES - SAVE
 BRA POMAC2
*
POMAC5 CLR LBFLG CLEAR LABEL FLAG
 LDX CULINE POINT TO LINE
 LDAA 0,X
 CMPA #$20 SPACE?
 BEQ POMAC6 YES
 JSR NXTOK GET LABEL
 COM LBFLG SET LABEL FLAG
POMAC6 JSR NXTOK GET MNEMONIC
 LDAA #4 SET FOR COMPARE
 STAA PCOUNT
 LDX DESCRA
 STX PSTNG1
 LDX #MEND
 STX PSTNG2
 LDX #PSTNG1 POINT TO PARAMS
 JSR COMPAR
 BNE POMAC8
 TST LBFLG LABELED?
 BEQ POMAC7 YES
 LDX #$0227 ERROR
 JSR PRINTE
 BRA POMAC7
*
POMAC8 JSR MACMOV PUT INTO MACTBL
 JMP POMAC2
*
POMAC7 TST PASS
 BNE POMACA PASS2
 LDAA #$17 ETB TO END OF MACRO
 LDX MACPTR
 STAA 0,X
 INX
 STX MACPTR
POMACA JMP MAIN1 ALL DONE

*
THE STACK

 BEQ PSPLER YES

*	STACK
*
PULIF STS STKSAV

 LDS IFSTKP

*
POPAG JSR ADRINT

* THIS WILL OPEN A LIBRARY FILE IF THERE'S
* NOT ONE OPEN YET.
*
POLIB TST LIBFLG
 BNE PLIBE1 ALLOW ONLY ONE LEVEL
 INC LIBFLG SET FLAG
 LDX CUCHAR POINT TO FILE SPEC
 STX BUFPNT PUT IN FLEX PLACE
 LDX #LIBFCB POINT TO LIB FCB
 JSR GETFIL PUT IN FILE SPEC IN FCB
 BCS PLIBE2 FILE SPEC ERROR
 LDAA #$01 DEFAULT = TEXT
 LDX #LIBFCB
 JSR SETEXT
 LDX #LIBFCB
 LDAA #$01 OPEN FUNCTION (READ)
 STAA 0,X
 JSR FMS OPEN IT
 BNE PLIBE3 ERROR ON OPEN
 JSR ADRINT DO THIS STUFF
 BRA POLIBE EXIT
*
PLIBE1 LDX #$0400 LIBE ERROR CODES
 BRA PLIBER
PLIBE2 LDX #$0401
 BRA PLIBER
PLIBE3 LDX #$0402
PLIBER JSR PRINTE
 CLR LIBFLG
POLIBE JMP MAIN1
