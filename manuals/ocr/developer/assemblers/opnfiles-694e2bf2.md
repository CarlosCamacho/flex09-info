* THIS SEGMENT WILL OPEN THE SOURCE FILE
* FOR READ AND A BINARY FILE FOR WRITE
*
OPNFLS LDX #RFCB POINT TO READ FCB
 JSR GETFIL PUT FILE SPEC IN FCB
 BCS ILLFS ILLEGAL FILE SPEC
 LDX #RFCB
 LDAA #$01 SET "TXT" EXT
 JSR SETEXT
 LDX #RFCB+3
 STX FROM
 LDX #WFCB+3
MOVRTW STX TO MOVE READ FILE NAME
 LDX FROM .TO WRITE FILE NAME
 LDAA 0,X
 INX
 STX FROM
 LDX TO
 STAA 0,X
 INX
 CPX #WFCB+15
 BNE MOVRTW DO TILL DONE
*
 CLR WFCB+12 KILL TXT EXT IN WFCB
 LDX #RFCB
 LDAA #$01 OPEN FOR READ
 STAA 0,X
 JSR FMS DO IT
 BNE OPNERR ERROR ON READ OPEN
 JSR TSTTRM TEST TERMINATOR
 BEQ NOTPLS
 BMI GETOPT GET OPTIONS
 LDX #WFCB FOUND COMMA
 JSR GETFIL GET OUTPUT FILE SPEC
 BCC OPNOK
ILLFS LDX #IFSMSG ILLEGAL FILE SPEC
 JSR PSTRNG PRINT IT
 JMP WARMS
*
OPNOK JSR TSTTRM
 BEQ NOTPLS
 BPL NOTPLS
*
GETOPT JSR NXTCH GET NEXT CHAR FROM
 BCS NOTPLS
 LDX #OPTTBL .LINE BUFFER
 ANDA #$5F MAKE UPPER CASE
*
OPTLUP CMPA 0,X
 BEQ SETOPT FOUND ONE
 INX
 INX
 INX
 TST 0,X AT END OF TABLE?
 BNE OPTLUP NOT YET
 LDX #ILLOPT ILLEGAL OPTION
 JSR PSTRNG
 JMP WARMS
*
SETOPT INX
 LDX 0,X
 CLR 0,X
 BRA GETOPT
*
NOTPLS LDAA OBJFLG
 BEQ NOBJ
 LDX #WFCB
 LDAA #$00
 JSR SETEXT SET "BIN" EXT
 LDX #WFCB
 LDAA #$02 OPEN FOR WRITE
 STAA 0,X
 JSR FMS DO IT
 BNE OPNERR
 LDAA #$FF TURN OFF SPC CMPRS
 STAA 59,X
NOBJ TST NOLFLG
 BNE OBJRTS
 CLR NOGFLG
 CLR LINFLG
OBJRTS RTS
*
OPNERR LDAA 1,X GET ERROR CODE
 CMPA #$03 IS IT "FILE EXISTS"?
 BNE CHKNSF NO-CHK "NO SUCH FILE"
 LDAA YFLG YES=OK TO DELETE
 BEQ OKTDEL THEN DO IT
 PSHA
 LDAA #$01
 STAA OUTSWT
 PULA
 LDX #DELMSG
 JSR PSTRNG
 JSR INCH
 CLR OUTSWT
 ANDA #$5F MAKE UPPER CASE
 CMPA #'Y
 BNE EXIT1
*
OKTDEL LDX #WFCB
 LDAA #$0C DEL EXISTING FILE
 STAA 0,X
 JSR FMS DELETE IT
 BNE GIVUP "GIVE IT UP"
 LDAA 36,X
 STAA 4,X FIX FIRST CHAR
 BRA NOTPLS
*
CHKNSF CMPA #$04
 BNE GIVUP
 LDX #NSFMSG "NO SUCH FILE"
 JSR PSTRNG
 BRA EXIT2
*
GIVUP JSR RPTERR
EXIT1 JSR FMSCLS
EXIT2 JMP WARMS
*
TSTTRM LDAA LSTTRM GET LAST TERMINATOR
 CMPA #$0D WAS IT CARET
 BEQ TT4 YES
 CMPA DEOL WAS IT "EOL"?
 BEQ TT4 YES
 CMPA #'+ WAS IT PLUS?
 BNE TT2 NO
TT1 LDAA #$FF
 BRA TT4
TT2 LDX BUFPNT
 LDAA 0,X
 INX
 CMPA #'+
 BNE TT3
 STX BUFPNT (FLEX'S POINTER)
 BRA TT1
TT3 LDAA #$01
TT4 RTS
*
