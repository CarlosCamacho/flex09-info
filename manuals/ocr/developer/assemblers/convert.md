*
* THESE ROUTINES CONVERT HEX, ASCII, AND DECIMAL
* TO BINARY FOR THE EVALUATE ROUTINE.
*
CVHB LDX DESCRA GET ADDRESS OF STRING
 CLR HVAL
 CLR HVAL+1
 LDAB DESCRC GET COUNT
 DEX
CVHB1 INX
 DECB
 BNE CVHB1
 LDAB DESCRC GET COUNT AGAIN
 JSR CVHBS CONVERT
 STAA HVAL+1
 DECB
 BEQ CVHBD
 DEX
 JSR CVHBS CONVERT
 ASLA
 ASLA
 ASLA
 ASLA
 ORAA HVAL+1
 STAA HVAL+1
 DECB
 BEQ CVHBD
 DEX
 JSR CVHBS CONVERT
 STAA HVAL
 DECB
 BEQ CVHBD
 DEX
 JSR CVHBS CONVERT
 ASLA
 ASLA
 ASLA
 ASLA
 ORAA HVAL
 STAA HVAL
CVHBD LDX HVAL GET FINAL VALUE
 RTS
*
* ROUTINE TO CONVERT ASCII TO BINARY
*
CVHBS LDAA 0,X
 SUBA #$30
 CMPA #$09
 BLE *+4
 SUBA #$07
 RTS
*
* CONVERT DECIMAL TO BINARY
*
* ON ENTRY DESCRA = ADDRESS OF DEC STRING
*
DESCRC = # OF BYTES IN STRING
*
* ON EXIT	INDEX = VALUE IN BINARY
*
CVDB CLR DVAL
 CLR DVAL+1
 CLR TENVL
 CLR TENVL+1
 INC TENVL+1
 LDX DESCRA GET ADDRESS OF STRING
 DEX
 LDAB DESCRC GET STRING BYTE COUNT
 STAB DCOUNT INIT COUNTER
CVDB1 INX
 DECB
 BNE CVDB1 GET TO LS DIGIT
CVDB2 STX DXSAV SAVE POINTER
 LDAB 0,X
 ANDB #$0F STRIP UPPER DIGIT
 CLRA
 LDX #TENVL POINT TO POWER OF TEN
 JSR MPY16
 ADDB DVAL+1
 ADCA DVAL
 STAA DVAL
 STAB DVAL+1
 CLRA
 LDAB #10
 LDX #TENVL
 JSR MPY16
 STAA TENVL
 STAB TENVL+1
 LDX DXSAV
 DEX
 DEC DCOUNT
 BNE CVDB2
 LDX DVAL
 RTS
