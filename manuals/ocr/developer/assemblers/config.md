TTL CONFIG
 PAG
*
* THIS SECTION WILL CONFIGURE THE TABLES
* USED BY THE ASSEMBLER.
*
 LDX #TABLES
 STX MACTBL INIT MACRO TABLE ADDRESS
*
 STX PSTNG1
 LDX #$0800
 STX PSTNG2
 LDX #PSTNG1
 JSR ADD16
 LDX PSTNG1
 STX MACEND INIT MACRO END ADDRESS
*
 LDX #$0100
 STX PSTNG2
 LDX #PSTNG1
 JSR ADD16
 LDX PSTNG1
 STX MACSTK INIT MACRO STACK ADDRESS
*
 INX
 STX SYMTAB INIT SYMBOL TABLE ADDRESS
*
 STX PSTNG2
 LDX #$7FFF-START FOR 32K SYSTEM
 STX PSTNG1
 LDX #PSTNG1
 JSR SUB16
 LDAA PSTNG1
 LDAB PSTNG1+1
 LDX #$0009
 STX PSTNG2
 LDX #PSTNG2
 JSR DIV16
 STAA NSYM
 STAB NSYM+1 INIT NUMBER OF SYMBOLS
*
 LDX #$0009
 STX PSTNG1
 LDX #PSTNG1
 JSR MPY16
 STAA PSTNG1
 STAB PSTNG1+1
 LDX SYMTAB
 STX PSTNG2
 LDX #PSTNG1
 JSR ADD16
 LDX PSTNG1
 STX SYMEND
*
 LDAA #$20 BLANK OUT SYMBOL TABLE
 LDX SYMTAB POINT TO SYMTAB
 STAA 0,X
 INX INCR POINTER
 CPX SYMEND DONE?
 BNE *-6 NO.
*
 LDX #$0000
 STX TSTPH CLEAR TSTPH
 STX ECOUNT CLEAR ERROR COUNT
 STX CMNLC INIT COMMON LC
*
PASS2 JSR ADRINT CLEAR FLAGS
 CLR LCOUNT LCOUNT := 0
 LDX MACTBL INIT MACRO TABLE POINTER
 STX MACPTR
 CLR MACFLG MODE := NOT-MACRO
 LDX MACSTK INIT MACRO STACK POINTER
 STX MSTKPT

 LDX #IFSTK
