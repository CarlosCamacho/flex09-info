* EVALUATE AN EXPRESSION, NUMBERS AND SYMBOLS
*
NSEVL CLR VALUE
 CLR VALUE+1 VALUE := 0
 CLR CLFLG CLFLG := 0
 STAA CLASS SAVE CLASS OF TOKEN
 CMPB #'* IS IT ASTERISK?
 BNE NSVLC1 NO
*
* EXPRESSION = * (PROG CTR)
*
 LDX LC YES - GET PROG CTR
 STX VALUE VALUE := LC
 LDAA #2
 STAA CLFLG CLFLG := 2
 COM RELFLG RELFLG := RELOCATE
NSVLA LDX CUCHAR
 LDAA 0,X GET NEXT CHARACTER
 CMPA #$20 IS IT SPACE?
 BEQ NSVLB YES
 CMPA #$0D IS IT EOL?
 BEQ NSVLB YES
 CMPA #', IS IT A COMMA?
 BNE NSVLC NO
NSVLB LDX VALUE
 STX ADR1
 CLRB RC := 0
 RTS
*
NSVLC JSR NXTOK GET NEXT TOKEN
 STAA CLASS SAVE TOKEN CLASS
NSVLC1 CMPA CLFLG CLASS = CLFLG?
 BNE NSVLF NO
NSVLD LDX #$0204 ERROR NUMBER
NSVLE CLRB
 COMB RC := FF
 RTS
*
NSVLF CMPA #$02 IS CLASS = STRING?
 BEQ NSVLH YES
 CMPA #$24 IS IT ARITHMETIC?
 BNE NSVLD NO - ERROR
NSVLG TST CLFLG IS IT 0?
 BEQ NSVLD YES - ERROR
 STAB OPERN SAVE OPERATOR
 STAA CLFLG SAVE CLASS FLAG
 JMP NSVLA SCAN AGAIN
*
NSVLH CMPB #$03 IS STRING HEX DATA?
 BNE NSVLJ NO
 LDAB DESCRC GET COUNT
 CMPB #4 IS COUNT > 4?
 BLE NSVLH1 NO - OK
 LDX #$0210 SET ERROR MESSAGE
 BRA NSVLE
NSVLH1 JSR CVHB CONVERT
 BRA NSVLM
*
NSVLJ CMPB #$09 IS STRING DECIMAL?
 BNE NSVLK NO

 BLE NSVLJ1 NO - OK
 LDX #$0210 SET ERROR MESSAGE
 BRA NSVLE
NSVLJ1 JSR CVDB CONVERT
 BRA NSVLM
