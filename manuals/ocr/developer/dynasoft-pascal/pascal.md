OPT PAG
 NAM PASCAL.CMD FOR DYNASOFT PASCAL 1.3
 PAG
*	COMMAND FILE FOR DYNASOFT PASCAL COMPILER
*	VERSION 1.3	MAY 10, 1981
*	AUTHOR ALLAN G. JOST
*
*	(C) 1981, DYNASOFT SYSTEMS LIMITED
*
*
*	SYNTAX: PASCAL,<INPUT FILE SPEC>,<BINARY FILE SPEC>
*	INPUT EXTENSION DEFAULTS TO .TXT
*	OUTPUT EXTENSION FORCED TO	.PCD
*	OLD OUTPUT ALWAYS DELETED
*
*	LOADS INTERPRETER AND COMPILER AND THEN
*	OPENS INPUT FILE AND REDIRECTS INPUT (CH 0)
*
*	EXTERNAL EQUATES
WARMS EQU $CD03
GETCHR EQU $CD15
PUTCHR EQU $CD18
NXTCH EQU $CD27
PSTRNG EQU $CD1E
PCRLF EQU $CD24
GETFIL EQU $CD2D
SETEXT EQU $CD33
FMSCLO EQU $D403
FMS EQU $D406
FCB EQU $C840
MEMEND EQU $CC2B
EOLN EQU $CC02
RPTERR EQU $CD3F
LSTERM EQU $CC11
COMP EQU $0100 COMPILER LOCATION
WS EQU $0020
EOI EQU $0022
TMP EQU $0030
FMSP EQU $0036 POINTER TO FLEX FMS
INTER EQU $0100 INTERPRETER ENTRY
*
 ORG $C100 MAIN ENTRY
COMPIL BRA COMP1
 FCB $60 VERSION NUMBER
STDIN FDB GETCHR FCB FOR KEYBOARD
 FCB 3 INPUT DEVICE
STDOUT FDB PUTC FCB FOR DISPLAY
 FCB 4 OUTPUT DEVICE
SAVNAM RMB 12 SPACE TO SAVE FILE SPEC
*	SUBROUTINE TO SET UP DEFAULT I/O
STDIO LDX #STDIN DEFAULT FCB'S
 STX CHAN0
 LDX #STDOUT
 STX CHAN0+2
 STX CHAN0+4
 LDX #FMS SET UP FMS VECTOR
 STX FMSP
 RTS
*
*
*	SUBROUTINE TO LOAD A RELOCATABLE FILE
*	ENTER WITH Y->FILE SPEC, U->LOAD POINT
*	RETURN WITH U->FIRST BYTE AFTER LOAD
*
LOADBIN PSHS U SAVE U
 LDX #FCB
 LEAU 3,X LOCATION OF FILE SPEC IN FCB
 BSR MOVNAM MOVE SPEC IN
 PULS U RESTORE U
*	ENTER HERE IF FILESPEC ALREADY IN FCB
LOADIT LDD #$01FF OPEN READ, BINARY FILE
 BSR OPEN OPEN THE FILE
 STU WS
 JSR FMS GET BYTE COUNT
 BNE ERROR
 TFR A,B
 JSR FMS
 BNE ERROR
 EXG A,B
 ADDD WS COMPUTE LAST BYTE+1
 STD EOI
RLOOP JSR FMS GET A BYTE
 BNE ERROR
 STA ,U+
 CMPU EOI LAST BYTE?
 BLO RLOOP NOT YET
*	ENTER HERE TO CLOSE A FILE
CLOSE LDD #$0400
*	ENTER HERE TO OPEN A FILE (MUST SET UP A,B)
OPEN STA ,X
 JSR FMS
 BNE ERROR
 STB 59,X SET COMPRESSION FLAG
 RTS
*	SUBROUTINE TO MOVE A FILE SPEC
MOVNAM LDB #12 NUMBER OF BYTES
MOV1 LDA ,Y+
 STA ,U+
 DECB
 BNE MOV1
 RTS
*
ERROR JSR RPTERR REPORT ERROR
NOSAVE JSR FMSCLO CLOSE ALL FILES
ALLDON JMP WARMS RETURN TO FLEX
*
COMP1 CLRA SET DP TO ZERO
 TFR A,DP
 BSR STDIO SET UP STANDARD I/O
 LDY #INTNAM INTEPRETER FILE
 LDU #INTERP LOCATION
 BSR LOADBIN LOAD INTERPRETER
LDCOMP LDY #COMPNAM COMPILER NAME
 LDU #COMP LOAD AT $0100
 BSR LOADBIN LOAD COMPILER
 STU WS WORKSPACE AFTER COMPILER
 JSR GETFIL GET INPUT FILENAME
 BCS ERROR
 LDA #1 TXT EXTENSION
 JSR SETEXT
 LDD #$0100 OPEN TEXTFILE
 BSR OPEN
 STX CHAN0 REDIRECT INPUT FOR COMPILER
 LDU WS WORKSPACE
 LEAU 8192,U 8K SIZE
 STU EOI NOTE STACK WILL START HERE
 LDY #COMP COMPILER LOCATION
 LDD MEMEND HEAP
 LDX #CHAN0 I/O TABLE
 LBSR INTERP EXECUTE THE COMPILER
 BPL NOSAVE IF COMPILER BOMBED
 LDX #FCB
 BSR CLOSE CLOSE THE FILE
*	NOW SAVE THE P-CODE
 LDA LSTERM CHECK LAST TERMINATOR
 CMPA #$0D CR?
 BEQ PCD USE OLD NAME
 CMPA EOLN
 BEQ PCD
 JSR GETFIL ELSE GET OUTPUT FILE SPEC
 BCS ERROR
PCD LDD #$5043 FORCE .PCD EXTENSION
 STD 12,X
 LDA #'D
 STA 14,X
 LEAY 3,X PREPARE TO SAVE FILE SPEC
 LDU #SAVNAM
 LBSR MOVNAM
 LDA #$0C DELETE CODE
 STA ,X
 JSR FMS DELETE THE .PCD FILE
 BEQ DELOK
 LDA 1,X CHECK ERROR
 CMPA #4 THIS ONE OK
 BNE ERROR2 ABORT ON BAD ONE
DELOK LDY #SAVNAM RESTORE FILE SPEC
 LEAU 3,X
 LBSR MOVNAM
*	NOW SAVE THE FILE
 LDD #$02FF OPEN BINARY FOR WRITE
 LBSR OPEN
 LDD EOI COMPUTE BYTE COUNT
 SUBD WS
 ADDD #1
 JSR FMS WRITE BYTE COUNT
 BNE ERROR2
 TFR B,A
 JSR FMS
 BNE ERROR2
 LDU WS
WLOOP CMPU EOI
 BHI WDONE NO MORE TO WRITE
 LDA ,U+ PICK UP A BYTE
 JSR FMS WRITE IT
 BEQ WLOOP
ERROR2 LBRA ERROR IF WRITE ERROR
WDONE LBSR CLOSE CLOSE THE FILE
 LBRA ALLDON
*
*	INTERFACE TO FLEX PUTCHR, USES PCRLF
PUTC CMPA #$0D CR?
 BEQ WRLN
 JMP PUTCHR FOR MOST CHARS
WRLN JMP PCRLF FOR END OF LINE
*
*	FILE SPECS
INTNAM FCC $FF,'INTERP',0,0,'SYS'
COMPNAM FCC $FF,'PASCAL',0,0,'SYS'
*
*	I/O TABLE
CHAN0 RMB 6 ROOM FOR 3 CHANNELS
INTERP EQU * INTERPRETER WILL LOAD HERE
*
*
 END COMPIL
