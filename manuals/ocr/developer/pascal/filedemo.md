PROGRAM FILES ( CHAFILE,BYTFILE,INTFILE ) ;
VAR	(* COMMAND IS RUN FILEDEMO A B C *)
	CHAFILE : FILE OF CHAR ;
	BYTFILE : FILE OF BYTE ;
	INTFILE : FILE OF INTEGER ;
	J	: INTEGER ; CH : CHAR ; B : BYTE ;
BEGIN

	REWRITE(CHAFILE); (* OPEN FOR WRITE *)
	
	FOR J:= ORD("A") TO ORD("Z") DO WRITE(CHAFILE,CHR(J));

	RESET(CHAFILE); (* OPEN FOR READ *)

	REPEAT
	READ(CHAFILE,CH);
	IF NOT EOF(CHAFILE) THEN WRITE(CH);
	UNTIL EOF(CHAFILE);

	WRITELN;
	REWRITE(BYTFILE); REWRITE(INTFILE);

	(* VARIABLES DECLARED TO BE OF TYPE BYTE
	MAY BE WRITTEN ON TO INTEGER FILES,
	AND INTEGERS MAY BE WRITTEN ON TO BYTE FILES,
	PROVIDED THAT THE INTEGERS ARE IN THE RANGE 0..255
	OR A RUN-TIME ERROR WILL RESULT.
	BYTE TO INTEGER OR INTEGER TO BYTE CONVERSION
	IS DONE AUTOMATICALLY BY THE COMPILER.
	THE SAME APPLIES TO READ OPERATIONS. *)
	
	FOR J:=0 TO 255 DO
	BEGIN
	WRITE(BYTFILE,J); WRITE(INTFILE,J);
	END;
	
	WRITELN;
	RESET(BYTFILE); RESET(INTFILE);

	REPEAT
	WRITELN;
	READ(BYTFILE,B); READ(INTFILE,J); WRITE(B,J);
	UNTIL J=255 ; (* TEST FOR A CONDITION
	OTHER THAN END OF FILE
	ON BINARY ( BYTE OR INTEGER ) FILES	
	SINCE FLEX ONLY SETS EOF TO TRUE WHEN
	THE FILE HAS BEEN READ TO THE END
	OF ITS LAST SECTOR *)
	WRITELN;
	RESET(BYTFILE); RESET(INTFILE);

	REPEAT
	WRITELN;
	READ(BYTFILE,J); READ(INTFILE,B); WRITE(B,J);
	UNTIL J=255 ;
	(* RE-OPEN THESE FILES FOR WRITE,
	IN ORDER TO RETURN THE SECTORS USED
	OTHERWISE THEY WILL BE CATALOG'D *)
	REWRITE(CHAFILE);
	REWRITE(BYTFILE);
	REWRITE(INTFILE);
END.
