GTYPE LDX TYPEE GET TYPE AND RANGE
 STAA $00,X
 INX
 BSR GADRS GET RANGE
 BLS GOTYP
BADRG JSR XCRLF BAD RANGE
 LDX #INVRG
 JSR XPDAT
 JSR STOPIN RESET GET FLAG
 JSR XCRLF
 JMP ASKIN EXIT ON ERROR
GOTYP LDAA COMAND CHK FOR M T
 CMPA #$4D M
 BEQ GOTYX
 JSR XINEE END OF LINE
GOTYX LDX EADDR UPDATE POINTER
 CLR $00,X CLEAR MAP LOCNS
 INX
 CLR $00,X
 INX
 CLR $00,X
 INX
 STX TYPEE
 RTS

GADRS STX EADDR GET ADDR RANGE
 LDX #LSTAR PRINT START=
 JSR XPDAT
 JSR XBADR READ LOW LIMIT
 LDX EADDR
 LDAA BADDR
 LDAB BADDR+1
 STAA $00,X
 INX
 STAB $00,X
 INX
 STX EADDR
 LDX #LENDR PRINT END=
 JSR XPDAT
 JSR XBADR READ HIGH LIMIT
 LDX EADDR
 LDAA BADDR
 CMPA MAXMEM CHK ADDR > MAX MEMORY
 BHI GADDX
 LDAB BADDR+1
 STAA $00,X
 INX
 STAB $00,X
 INX
 STX EADDR
 DEX CHK NEGATIVE RANGE
 DEX
 DEX
 DEX
 LDAA $00,X
 CMPA $02,X
 BNE GADDX
 LDAA $01,X
 CMPA $03,X
GADDX RTS

RDLINE STX RADDR GET INPUT LINE
 JSR INBUFR
 LDX BUFPTR
 LDX $00,X
 STX SAVEX
 LDAB #$1F
RDLINM LDX SAVEX TRANSFER LINE
 LDAA $00,X
 INX
 STX SAVEX
 LDX RADDR
 ANDA #$7F
 CMPA #$20 CHK CTRL
 BMI RDLINX
 STAA $00,X
 INX

 DECB
 BNE RDLINM
RDLINX LDAA #$04 PUT EOT AFTER INPUT
